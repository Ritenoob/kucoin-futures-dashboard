const WebSocket = require('ws');
const fs = require('fs');
const path = require('path');
const { promisify } = require('util');
const writeFile = promisify(fs.writeFile);

const CONFIG = {
  symbols: ['BTC-USDT', 'ETH-USDT'],
  wsUrl: 'wss://ws-api-spot.kucoin.com/',
  restUrl: 'https://api.kucoin.com',
  candleInterval: 60000, // 1 minute
  outputDir: path.join(__dirname, '../data/live'),
  logFile: path.join(__dirname, '../logs/collector.log')
};

// Ensure directories exist
[CONFIG.outputDir, path.dirname(CONFIG.logFile)].forEach(dir => {
  if (!fs.existsSync(dir)) {
    fs.mkdirSync(dir, { recursive: true });
  }
});

class KuCoinCollector {
  constructor() {
    this.ws = null;
    this.currentCandles = {}; // Symbol -> current candle
    this.reconnectAttempts = 0;
    this.maxReconnectAttempts = 10;
    this.reconnectDelay = 5000;
    this.candleTimer = null;
    
    // Output streams
    this.outputFile = path.join(CONFIG.outputDir, 'market_data.jsonl');
    this.writeStream = fs.createWriteStream(this.outputFile, { flags: 'a' });
    
    this.log('✓ Collector initialized');
  }
  
  log(message) {
    const timestamp = new Date().toISOString();
    const logMessage = `[${timestamp}] ${message}\n`;
    console.log(message);
    fs.appendFileSync(CONFIG.logFile, logMessage);
  }
  
  async getWebSocketToken() {
    try {
      const response = await fetch(`${CONFIG.restUrl}/api/v1/bullet-public`);
      const data = await response.json();
      
      if (data.code !== '200000') {
        throw new Error(`Failed to get token: ${data.msg}`);
      }
      
      return data.data;
    } catch (error) {
      this.log(`✗ Error getting WebSocket token: ${error.message}`);
      throw error;
    }
  }
  
  async connect() {
    try {
      this.log('Connecting to KuCoin WebSocket...');
      
      const tokenData = await this.getWebSocketToken();
      const endpoint = tokenData.instanceServers[0].endpoint;
      const token = tokenData.token;
      const connectId = Date.now();
      
      const wsUrl = `${endpoint}?token=${token}&connectId=${connectId}`;
      this.ws = new WebSocket(wsUrl);
      
      this.ws.on('open', () => this.onOpen());
      this.ws.on('message', (data) => this.onMessage(data));
      this.ws.on('error', (error) => this.onError(error));
      this.ws.on('close', () => this.onClose());
      
      // Ping/pong to keep connection alive
      this.pingInterval = setInterval(() => {
        if (this.ws && this.ws.readyState === WebSocket.OPEN) {
          this.ws.send(JSON.stringify({ type: 'ping', id: Date.now() }));
        }
      }, 20000);
      
    } catch (error) {
      this.log(`✗ Connection error: ${error.message}`);
      this.scheduleReconnect();
    }
  }
  
  onOpen() {
    this.log('✓ Connected to KuCoin WebSocket');
    this.reconnectAttempts = 0;
    this.subscribe();
    this.startCandleAggregation();
  }
  
  subscribe() {
    CONFIG.symbols.forEach(symbol => {
      const subscribeMsg = {
        id: Date.now(),
        type: 'subscribe',
        topic: `/market/ticker:${symbol}`,
        privateChannel: false,
        response: true
      };
      
      this.ws.send(JSON.stringify(subscribeMsg));
      this.log(`✓ Subscribed to ${symbol}`);
      
      // Initialize candle for this symbol
      this.currentCandles[symbol] = null;
    });
  }
  
  onMessage(data) {
    try {
      const msg = JSON.parse(data);
      
      if (msg.type === 'message' && msg.topic && msg.topic.includes('/market/ticker')) {
        this.handleTicker(msg);
      } else if (msg.type === 'pong') {
        // Keep-alive acknowledged
      } else if (msg.type === 'ack') {
        // Subscription acknowledged
      }
    } catch (error) {
      this.log(`✗ Error parsing message: ${error.message}`);
    }
  }
  
  handleTicker(msg) {
    const symbol = msg.subject;
    const tick = msg.data;
    const price = parseFloat(tick.price);
    const volume = parseFloat(tick.vol);
    const timestamp = parseInt(tick.time);
    
    // Initialize or update current candle
    if (!this.currentCandles[symbol]) {
      this.currentCandles[symbol] = {
        symbol,
        timestamp,
        open: price,
        high: price,
        low: price,
        close: price,
        volume: 0,
        trades: 0
      };
    }
    
    const candle = this.currentCandles[symbol];
    candle.high = Math.max(candle.high, price);
    candle.low = Math.min(candle.low, price);
    candle.close = price;
    candle.volume += volume;
    candle.trades += 1;
  }
  
  startCandleAggregation() {
    // Emit candles every minute
    this.candleTimer = setInterval(() => {
      this.emitCandles();
    }, CONFIG.candleInterval);
    
    this.log('✓ Candle aggregation started (1 minute intervals)');
  }
  
  emitCandles() {
    const now = Date.now();
    
    Object.keys(this.currentCandles).forEach(symbol => {
      const candle = this.currentCandles[symbol];
      
      if (candle && candle.trades > 0) {
        // Round timestamp to minute
        candle.timestamp = Math.floor(now / 60000) * 60000;
        
        // Write to file
        const line = JSON.stringify(candle) + '\n';
        this.writeStream.write(line);
        
        this.log(`📊 ${symbol}: O:${candle.open} H:${candle.high} L:${candle.low} C:${candle.close} V:${candle.volume.toFixed(2)}`);
        
        // Reset candle for next period
        this.currentCandles[symbol] = null;
      }
    });
  }
  
  onError(error) {
    this.log(`✗ WebSocket error: ${error.message}`);
  }
  
  onClose() {
    this.log('✗ WebSocket connection closed');
    
    if (this.pingInterval) {
      clearInterval(this.pingInterval);
    }
    
    if (this.candleTimer) {
      clearInterval(this.candleTimer);
    }
    
    this.scheduleReconnect();
  }
  
  scheduleReconnect() {
    if (this.reconnectAttempts >= this.maxReconnectAttempts) {
      this.log(`✗ Max reconnection attempts (${this.maxReconnectAttempts}) reached. Exiting.`);
      process.exit(1);
    }
    
    this.reconnectAttempts++;
    const delay = this.reconnectDelay * this.reconnectAttempts;
    
    this.log(`⏳ Reconnecting in ${delay/1000}s (attempt ${this.reconnectAttempts}/${this.maxReconnectAttempts})...`);
    
    setTimeout(() => {
      this.connect();
    }, delay);
  }
  
  shutdown() {
    this.log('Shutting down collector...');
    
    if (this.pingInterval) {
      clearInterval(this.pingInterval);
    }
    
    if (this.candleTimer) {
      clearInterval(this.candleTimer);
      this.emitCandles(); // Emit final candles
    }
    
    if (this.ws) {
      this.ws.close();
    }
    
    if (this.writeStream) {
      this.writeStream.end();
    }
    
    this.log('✓ Collector shut down gracefully');
  }
}

python"""
Real-time stream processor with production features:
- Rolling window buffer
- Error handling and logging
- Performance monitoring
- Graceful shutdown
"""

import json
import time
import signal
import sys
from pathlib import Path
from collections import deque
from datetime import datetime
import pandas as pd
import logging

from signal_engine import SignalEngine

# Configure logging
logging.basicConfig(
    level=logging.INFO,
    format='[%(asctime)s] %(levelname)s: %(message)s',
    handlers=[
        logging.FileHandler('logs/engine.log'),
        logging.StreamHandler(sys.stdout)
    ]
)
logger = logging.getLogger(__name__)


class StreamProcessor:
    def __init__(self, 
                 input_file='../data/live/market_data.jsonl',
                 output_file='../data/signals/signals.jsonl',
                 csv_file='../data/signals/signals.csv',
                 lookback=500):
        
        self.input_file = Path(input_file)
        self.output_file = Path(output_file)
        self.csv_file = Path(csv_file)
        self.lookback = lookback
        
        # Initialize signal engine
        logger.info("Initializing signal engine...")
        self.engine = SignalEngine()
        
        # Rolling data buffer per symbol
        self.buffers = {}
        
        # Track file positions
        self.last_position = 0
        
        # Performance metrics
        self.signals_generated = 0
        self.bars_processed = 0
        self.start_time = time.time()
        
        # Ensure directories exist
        self.output_file.parent.mkdir(parents=True, exist_ok=True)
        self.csv_file.parent.mkdir(parents=True, exist_ok=True)
        Path('logs').mkdir(exist_ok=True)
        
        # CSV header
        if not self.csv_file.exists():
            self.init_csv()
        
        # Shutdown flag
        self.running = True
        
        logger.info("✓ Stream processor initialized")
        logger.info(f"  Input: {self.input_file}")
        logger.info(f"  Output: {self.output_file}")
        logger.info(f"  CSV: {self.csv_file}")
        logger.info(f"  Lookback: {self.lookback} bars")
    
    def init_csv(self):
        """Initialize CSV with headers"""
        headers = [
            'timestamp', 'symbol', 'signal', 'strength', 'price',
            'volatility', 'AO', 'AOz', 'dAOz', 'AO_score', 'AO_state',
            'rsi', 'kdj_k', 'kdj_d', 'wr', 'ema50_tf', 'atr14'
        ]
        with open(self.csv_file, 'w') as f:
            f.write(','.join(headers) + '\n')
        logger.info(f"✓ Initialized CSV: {self.csv_file}")
    
    def get_buffer(self, symbol):
        """Get or create buffer for symbol"""
        if symbol not in self.buffers:
            self.buffers[symbol] = {
                'timestamp': deque(maxlen=self.lookback),
                'open': deque(maxlen=self.lookback),
                'high': deque(maxlen=self.lookback),
                'low': deque(maxlen=self.lookback),
                'close': deque(maxlen=self.lookback),
                'volume': deque(maxlen=self.lookback)
            }
        return self.buffers[symbol]
    
    def read_new_candles(self):
        """Read new candles from input file"""
        if not self.input_file.exists():
            return []
        
        try:
            with open(self.input_file, 'r') as f:
                f.seek(self.last_position)
                new_lines = f.readlines()
                self.last_position = f.tell()
            
            candles = []
            for line in new_lines:
                if line.strip():
                    try:
                        candle = json.loads(line)
                        candles.append(candle)
                    except json.JSONDecodeError as e:
                        logger.warning(f"Invalid JSON: {e}")
            
            return candles
        
        except Exception as e:
            logger.error(f"Error reading candles: {e}")
            return []
    
    def update_buffer(self, candle):
        """Add candle to appropriate buffer"""
        symbol = candle['symbol']
        buffer = self.get_buffer(symbol)
        
        buffer['timestamp'].append(pd.Timestamp(candle['timestamp'], unit='ms'))
        buffer['open'].append(float(candle['open']))
        buffer['high'].append(float(candle['high']))
        buffer['low'].append(float(candle['low']))
        buffer['close'].append(float(candle['close']))
        buffer['volume'].append(float(candle['volume']))
        
        self.bars_processed += 1
    
    def process_symbol(self, symbol):
        """Generate signal for symbol"""
        buffer = self.buffers.get(symbol)
        
        if not buffer or len(buffer['close']) < 100:
            return None
        
        try:
            # Convert buffer to DataFrame
            df = pd.DataFrame({
                'timestamp': list(buffer['timestamp']),
                'open': list(buffer['open']),
                'high': list(buffer['high']),
                'low': list(buffer['low']),
                'close': list(buffer['close']),
                'volume': list(buffer['volume'])
            })
            
            # Generate signals
            result = self.engine.generate_signals(df)
            
            # Get last bar
            last_row = result.iloc[-1]
            
            if pd.notna(last_row.get('signal')):
                signal = {
                    'timestamp': int(last_row.name.timestamp() * 1000),
                    'symbol': symbol,
                    'signal': last_row['signal'],
                    'strength': int(last_row['strength']),
                    'price': float(df.iloc[-1]['close']),
                    'indicators': {
                        'ao': float(last_row['AO']),
                        'aoz': float(last_row['AOz']),
                        'daoz': float(last_row['dAOz']),
                        'ao_state': last_row['AO_state'],
                        'rsi': float(last_row['rsi']),
                        'kdj_k': float(last_row['kdj_k']),
                        'kdj_d': float(last_row['kdj_d']),
                        'wr': float(last_row['wr']),
                        'volatility': float(last_row['volatility']),
                        'ema50_tf': float(last_row['ema50_tf']),
                        'atr14': float(last_row['atr14'])
                    }
                }
                return signal
        
        except Exception as e:
            logger.error(f"Error processing {symbol}: {e}")
        
        return None
    
    def write_signal(self, signal):
        """Write signal to JSONL and CSV"""
        try:
            # JSONL
            with open(self.output_file, 'a') as f:
                f.write(json.dumps(signal) + '\n')
            
            # CSV
            ind = signal['indicators']
            csv_row = [
                signal['timestamp'],
                signal['symbol'],
                signal['signal'],
                signal['strength'],
                signal['price'],
                ind['volatility'],
                ind['ao'],
                ind['aoz'],
                ind['daoz'],
                0,  # ao_score placeholder
                ind['ao_state'],
                ind['rsi'],
                ind['kdj_k'],
                ind['kdj_d'],
                ind['wr'],
                ind['ema50_tf'],
                ind['atr14']
            ]
            
            with open(self.csv_file, 'a') as f:
                f.write(','.join(map(str, csv_row)) + '\n')
            
            self.signals_generated += 1
            
            logger.info(
                f"🚨 {signal['symbol']} {signal['signal']} @ ${signal['price']:.2f} "
                f"(strength: {signal['strength']}) "
                f"[RSI: {ind['rsi']:.1f}, WR: {ind['wr']:.1f}]"
            )
        
        except Exception as e:
            logger.error(f"Error writing signal: {e}")
    
    def print_stats(self):
        """Print performance statistics"""
        elapsed = time.time() - self.start_time
        bars_per_sec = self.bars_processed / elapsed if elapsed > 0 else 0
        
        logger.info(
            f"📊 Stats: {self.bars_processed} bars processed, "
            f"{self.signals_generated} signals, "
            f"{bars_per_sec:.1f} bars/sec, "
            f"uptime: {elapsed/60:.1f}min"
        )
    
    def run(self, poll_interval=1, stats_interval=60):
        """Main processing loop"""
        logger.info("▶️  Starting stream processor...")
        logger.info(f"Polling interval: {poll_interval}s")
        
        last_stats_time = time.time()
        
        try:
            while self.running:
                # Read new candles
                candles = self.read_new_candles()
                
                if candles:
                    # Update buffers
                    for candle in candles:
                        self.update_buffer(candle)
                    
                    # Process each symbol
                    for symbol in self.buffers.keys():
                        signal = self.process_symbol(symbol)
                        if signal:
                            self.write_signal(signal)
                
                # Print stats periodically
                if time.time() - last_stats_time >= stats_interval:
                    self.print_stats()
                    last_stats_time = time.time()
                
                # Wait before next poll
                time.sleep(poll_interval)
        
        except KeyboardInterrupt:
            logger.info("\n⏸️  Received interrupt signal")
        
        except Exception as e:
            logger.error(f"Fatal error: {e}", exc_info=True)
        
        finally:
            self.shutdown()
    
    def shutdown(self):
        """Graceful shutdown"""
        logger.info("Shutting down stream processor...")
        self.running = False
        self.print_stats()
        logger.info("✓ Stream processor stopped")


def signal_handler(signum, frame):
    """Handle shutdown signals"""
    logger.info(f"\nReceived signal {signum}")
    sys.exit(0)


if __name__ == '__main__':
    # Register signal handlers
    signal.signal(signal.SIGINT, signal_handler)
    signal.signal(signal.SIGTERM, signal_handler)
    
    # Start processor
    processor = StreamProcessor()
    processor.run(poll_interval=1, stats_interval=60)
